  function norm(s){return s.normalize('NFD').replace(/[̀-ͯ]/g,'').toLowerCase().replace(/[-'’]/g,' ');}
  function runSearch(){
    var q=sIn.value.trim(); sRes.innerHTML=''; if(q.length<1) return;
    var out=[], m=q.match(/^([a-z]{1,2})\s*(?:n°\s*)?(\d+)$/i);
    if(m){ var s=m[1].toUpperCase(), n=+m[2];
      pomFeats.concat(ctxFeats).forEach(function(f){var p=f.properties; if(p.s===s&&p.n===n) out.push({k:'Parcelle',t:cname(p.c)+' '+p.s+' '+p.n,ct:inAoc(p)?'AOC':'hors AOC',go:function(){selectParcel(f); map.flyToBounds(L.geoJSON(f).getBounds(),{padding:[90,90],maxZoom:18,duration:.8});}});}); }
    if(q.length>=2){ var nq=norm(q);
      D.chateaux.forEach(function(c){ if(norm(c.n).indexOf(nq)>=0) out.push({k:'Domaine',t:c.n,ct:c.r?RANK[c.r].short:(c.in?CFG.shortName:''),go:function(){selectChateau(c);}}); });
      Object.keys(lieuIdx).forEach(function(k){ var L_=lieuIdx[k]; if(norm(L_.n).indexOf(nq)>=0) out.push({k:'Lieu-dit',t:tc(L_.n),ct:cname(L_.c),go:function(){selectLieu(L_.n,L_.c);}}); });
    }
    out.slice(0,16).forEach(function(r){ var d=document.createElement('div'); d.className='res';
      d.innerHTML='<span class="k">'+r.k+'</span><span>'+esc(r.t)+'</span><span class="ct">'+r.ct+'</span>'; d.onclick=r.go; sRes.appendChild(d); });
    if(!out.length) sRes.innerHTML='<div class="res-empty">'+(m?'Aucune parcelle chargée ne correspond : zoomez sur la commune pour charger ses parcelles.':'Aucun résultat')+'</div>';
  }
  sIn.addEventListener('input',runSearch);
